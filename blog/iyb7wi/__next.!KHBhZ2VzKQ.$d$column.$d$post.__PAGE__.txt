/**
 * 波兰表达式/逆波兰表达式求解运算表达式

/**
 * 单链表的存储结构
 */
typedef struct LNode {
    string data; //数据域
    struct LNode *next; //指针域
}Lnode, *LinkList; //LinkList为指向结构体LNode的指针类型

/* 初始化链表 */
void InitList(LinkList &amp;L) {
    L = new LNode;
    L-&gt;next = NULL;
}

/* 打印 */
void TraverseList(LinkList &amp; L){
        LNode *p = new LNode;
        p = L-&gt;next;
        // cout &lt;&lt; &quot;此中缀表达式链表打印的结果为：&quot;;
        while (p != NULL)
        {
            cout &lt;&lt; p-&gt;data;
            p = p-&gt;next;
        }
        cout &lt;&lt; &quot;\n&quot;;
}

/* 逆置 */
void ReverseList(LinkList &amp;L) {
    LNode *p = L-&gt;next;             
    L-&gt;next = NULL;                
    while(p)                        
    {
        LNode *q = p-&gt;next;         
        p-&gt;next = L-&gt;next;          
        L-&gt;next = p;                
        p = q;                      
    }
}

/**
 * 初始化用户输入的链表
 */
void Center(LinkList &amp;L,string s) {
    InitList(L);
    LinkList p = L;
    string temp = &quot;&quot;;
    for (int i = 0; i &lt; s.length();i++){
        // 处理双位数字情况
        if (isdigit(s[i])) {
            // 该字符为数字
            temp = temp + s[i];
            if (!isdigit(s[i+1])) {
                // 下一个不是数字，而是字符，将temp后插
                LinkList node = new LNode;
                node-&gt;data = temp;
                node-&gt;next = NULL;
                p-&gt;next = node;
                p = node;
                // 将temp重置
                temp = &quot;&quot;;
                continue;
            }
            continue;
        }
        // 后插到L尾巴上
        LinkList node = new LNode;
        node-&gt;data = s[i];
        node-&gt;next = NULL;
        p-&gt;next = node;
        p = node;
    }    
}


/**
 * 将表达式转换为波兰表达式/逆波兰表达式
 * 第二个参数对逆波兰而言是左括号，第三个参数对逆波兰而言是右括号
 * 对波兰而言反过来
 */
void Transition(LinkList &amp;L, string l, string r){
    // 定义一个栈用来处理
    stack&lt;string&gt; stack;
    LinkList p = L-&gt;next;
    LinkList result = new LNode;
    InitList(result);
    LinkList result_a = result;

    while(p != NULL) {
        if (p-&gt;data == l) {
            stack.push(p-&gt;data);
        } else if(p-&gt;data == r) {
            while(stack.top() != l){
                LinkList temp = new LNode;
                temp-&gt;data = stack.top();
                temp-&gt;next = NULL;
                result_a-&gt;next = temp;
                result_a = temp;
                stack.pop();
            }
            if (stack.top() == l){
                stack.pop();
            }
        } else if(p-&gt;data == &quot;+&quot; || p-&gt;data == &quot;-&quot;) {
            if (stack.size() != 0) {
                if (stack.top() == &quot;*&quot; || stack.top() == &quot;/&quot;){
                    for (int i = 0; i &lt; stack.size();i++) {
                        if (stack.top() == l) {
                            break;
                        }
                        LinkList temp = new LNode;
                        temp-&gt;data = stack.top();
                        temp-&gt;next = NULL;
                        result_a-&gt;next = temp;
                        result_a = temp;
                        stack.pop();
                    }
                }
            }
            stack.push(p-&gt;data);
        } else if(p-&gt;data == &quot;*&quot; || p-&gt;data == &quot;/&quot;) {
            stack.push(p-&gt;data);
        } else {
            LinkList temp = new LNode;
            temp-&gt;data = p-&gt;data;
            temp-&gt;next = NULL;
            result_a-&gt;next = temp;
            result_a = temp;
        }
        p = p-&gt;next;
    }
    // TraverseList(result);
    for (int i = 0; i &lt; stack.size();i++) {
        LinkList temp = new LNode;
        temp-&gt;data = stack.top();
        temp-&gt;next = NULL;
        result_a-&gt;next = temp;
        result_a = temp;
        stack.pop();
    }
    // 上一个操作总是不能清空栈的最后一个元素
    LinkList temp = new LNode;
    temp-&gt;data = stack.top();
    temp-&gt;next = NULL;
    result_a-&gt;next = temp;
    result_a = temp;
    stack.pop();

    L = result;
}

    while(p != NULL) {
        if (p-&gt;data == &quot;+&quot;){
