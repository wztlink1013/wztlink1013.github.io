 //其中，T非空，1&lt;=pos&lt;=S.length

void get_next(HString, int next[])
{//求模式串T的next函数值并存入数组next
    int j = 1, t = 0;
    next[1] = 0;
    while(j &lt; T.length)
    {
        if(t == 0 || T.ch[j] == T.ch[t])
        {
            t++;
            j++;
            next[j] = t;
        }
        else
            t = next[t];
    }
}

void get_nextval(HString T, int nextval[])
{//求模式串T的next函数修正值并存入数组nextval
    int j = 1, t = 0;
    nextval[1] = 0;
    while(j &lt; T.length)
    {
        if(t == 0 || T.ch[j] == T.ch[t])
        {
            t++;
            j++;
            if(T.ch[j] != T.ch[t])
                nextval[j] = t;
            else
                nextval[j] = nextval[t];
        }
        else
            t = nextval[t];
    }
}
